#!/usr/bin/env python3
"""
preprocess.py — Stage 1 of the validation-study data pipeline.

Turns the three supplied synthetic CSV files into a clean, deduplicated,
chronologically ordered set of clinical episodes ready for summary generation.

WHAT IT FIXES (see Dataset_Evidence_Report_v1.md for the evidence behind each):
  1. Duplication  — 19 patients appear twice with byte-identical note sets.
                    69 rows collapse to 50 distinct patients / 50 episodes.
  2. Encoding     — mojibake ("Â·", "Â°") in all 69 bed_location values and
                    124 note texts, from a UTF-8/Latin-1 double-encoding.
  3. Broken field — the `ward` column is the literal string "ward" in all rows;
                    real setting is parsed out of `bed_location`.
  4. Dates        — timestamps are DD/MM/YYYY. Parsing without dayfirst=True
                    silently mangles them (e.g. 12/01/2026 -> 1 Dec).
  5. Chronology   — notes are ordered per episode; notes predating the
                    admission timestamp are flagged as pre-admission
                    (legitimate for the 21 booked admissions, which have
                    pre-operative outpatient notes).

OUTPUT
  out/episodes.json        one record per episode, notes ordered, metadata attached
  out/episode_index.csv    flat index for quick inspection
  out/pseudonyms.csv       stable patient pseudonyms for use in figures/screenshots

--------------------------------------------------------------------------
ATTRIBUTION: This file was written with AI assistance (Claude, Anthropic) and
reviewed/adapted by the author. Retained per City St George's MSc Project
Guidance, Appendix C items 7 and 9, which require source code to be identified
as automatically generated, re-used, library or original.
--------------------------------------------------------------------------
"""

from __future__ import annotations

import argparse
import json
import re
import sys
from pathlib import Path

import pandas as pd

try:
    import ftfy
except ImportError:  # graceful fallback if ftfy is unavailable
    ftfy = None


# --------------------------------------------------------------------------
# Text repair
# --------------------------------------------------------------------------

def fix_text(value):
    """Repair UTF-8/Latin-1 double-encoding ('Â·' -> '·', 'Â°' -> '°')."""
    if not isinstance(value, str):
        return value
    if ftfy is not None:
        return ftfy.fix_text(value)
    try:
        return value.encode("latin-1").decode("utf-8")
    except (UnicodeEncodeError, UnicodeDecodeError):
        return value


# --------------------------------------------------------------------------
# Setting derivation
# --------------------------------------------------------------------------

# bed_location looks like: "Neurology Ward · Bay 00 · Bed B00"
def parse_department(bed_location: str) -> str:
    if not isinstance(bed_location, str):
        return "Unknown"
    return bed_location.split("·")[0].strip()


SETTING_PATTERNS = [
    ("Paediatric",  r"paediatric|pediatric|children'?s"),
    ("Orthopaedic", r"orthopaedic|orthopedic|trauma and orth|t&o"),
    ("Respiratory", r"respiratory"),
    ("Neurology",   r"neurolog|neurosurg"),
    ("Acute/Assessment", r"acute medical|assessment unit|\bamu\b|\bsau\b|\bpau\b|observation"),
    ("Surgical",    r"surg"),
]


def classify_setting(department: str) -> list[str]:
    """A department can legitimately match more than one setting
    (e.g. 'Paediatric Orthopaedic Ward 3B')."""
    d = department.lower()
    tags = [name for name, pattern in SETTING_PATTERNS if re.search(pattern, d)]
    return tags or ["Other"]


def is_high_dependency(department: str) -> bool:
    return bool(re.search(r"\bhdu\b|high dependency|\bicu\b|intensive care|critical care",
                          department, re.I))


# --------------------------------------------------------------------------
# Age banding (PRSB-agnostic, used for cohort description only)
# --------------------------------------------------------------------------

AGE_BINS = [0, 1, 5, 12, 18, 40, 65, 75, 200]
AGE_LABELS = ["<1", "1-4", "5-11", "12-17", "18-39", "40-64", "65-74", "75+"]


# --------------------------------------------------------------------------
# Main
# --------------------------------------------------------------------------

def main() -> int:
    ap = argparse.ArgumentParser(description=__doc__,
                                 formatter_class=argparse.RawDescriptionHelpFormatter)
    ap.add_argument("--data", default="data", help="directory holding the three CSVs")
    ap.add_argument("--out", default="out", help="output directory")
    args = ap.parse_args()

    data_dir, out_dir = Path(args.data), Path(args.out)
    out_dir.mkdir(parents=True, exist_ok=True)

    # ---- load -------------------------------------------------------------
    patients = pd.read_csv(data_dir / "patients.csv")
    admissions = pd.read_csv(data_dir / "admissions.csv")
    notes = pd.read_csv(data_dir / "synthetic_clinical_notes.csv")

    raw_counts = dict(patients=len(patients), admissions=len(admissions), notes=len(notes))

    # ---- 1. deduplicate ---------------------------------------------------
    # patients.csv contains 19 exact duplicate rows.
    patients = patients.drop_duplicates(subset="person_id").reset_index(drop=True)

    # 19 patients have two admission_ids whose note sets are byte-identical.
    # Keep one admission per patient, chosen deterministically by sorted id
    # so that repeated runs of this script give identical output.
    admissions = (admissions.sort_values("admission_id")
                            .drop_duplicates(subset="patient_id")
                            .reset_index(drop=True))
    kept_admissions = set(admissions["admission_id"])
    notes = notes[notes["admission_id"].isin(kept_admissions)].copy()

    # ---- 2. repair encoding ----------------------------------------------
    notes["clean_note_text"] = notes["clean_note_text"].map(fix_text)
    notes["note_subject"] = notes["note_subject"].map(fix_text)
    admissions["bed_location"] = admissions["bed_location"].map(fix_text)

    # ---- 3. derive setting ------------------------------------------------
    admissions["department"] = admissions["bed_location"].map(parse_department)
    admissions["setting_tags"] = admissions["department"].map(classify_setting)
    admissions["high_dependency"] = admissions["department"].map(is_high_dependency)
    admissions = admissions.drop(columns=["ward"])  # literal "ward", carries no information

    # ---- 4. parse dates (dayfirst is essential) ---------------------------
    for col in ("creation_timestamp", "ingest_timestamp", "updt_dt_tm"):
        notes[col] = pd.to_datetime(notes[col], format="mixed", dayfirst=True)
    admissions["admission_timestamp"] = pd.to_datetime(
        admissions["admission_timestamp"], format="mixed", dayfirst=True)
    patients["date_of_birth"] = pd.to_datetime(
        patients["date_of_birth"], format="mixed", dayfirst=True)

    # ---- 5. order chronologically, flag pre-admission ---------------------
    notes = notes.sort_values(["admission_id", "creation_timestamp"]).reset_index(drop=True)
    notes["note_seq"] = notes.groupby("admission_id").cumcount() + 1

    adm_start = admissions.set_index("admission_id")["admission_timestamp"]
    notes["pre_admission"] = (
        notes["creation_timestamp"] < notes["admission_id"].map(adm_start)
    )

    # ---- pseudonyms for figures ------------------------------------------
    pseudo = {pid: f"PT-{i + 1:03d}" for i, pid in enumerate(sorted(patients["person_id"]))}
    pd.DataFrame({"person_id": list(pseudo), "pseudonym": list(pseudo.values())}) \
        .to_csv(out_dir / "pseudonyms.csv", index=False)

    # ---- assemble episodes ------------------------------------------------
    pat_by_id = patients.set_index("person_id")
    episodes, index_rows = [], []

    for adm in admissions.to_dict("records"):
        aid, pid = adm["admission_id"], adm["patient_id"]
        ep_notes = notes[notes["admission_id"] == aid]
        if ep_notes.empty:
            continue
        p = pat_by_id.loc[pid]

        note_records = [{
            "note_id": r["clinical_note_id"],
            "seq": int(r["note_seq"]),
            "timestamp": r["creation_timestamp"].isoformat(),
            "note_type": r["note_type"],
            "note_subject": r["note_subject"],
            "pre_admission": bool(r["pre_admission"]),
            "text": r["clean_note_text"],
        } for r in ep_notes.to_dict("records")]

        first, last = ep_notes["creation_timestamp"].min(), ep_notes["creation_timestamp"].max()

        episodes.append({
            "episode_id": aid,
            "patient": {
                "person_id": pid,
                "pseudonym": pseudo[pid],
                "full_name": p["full_name"],          # synthetic; suppress in figures
                "nhs_number": str(p["nhs_number"]),   # synthetic; suppress in figures
                "date_of_birth": p["date_of_birth"].strftime("%d/%m/%Y"),
                "age": int(p["age"]),
                "gender_identity": p["gender_identity"],
                "is_paediatric": bool(p["age"] < 18),
            },
            "admission": {
                "method": adm["admission_method"],
                "timestamp": adm["admission_timestamp"].isoformat(),
                "bed_location": adm["bed_location"],
                "department": adm["department"],
                "setting_tags": adm["setting_tags"],
                "high_dependency": bool(adm["high_dependency"]),
                "site_name": adm["site_name"],
            },
            "span": {
                "first_note": first.isoformat(),
                "last_note": last.isoformat(),
                "days": round((last - first).total_seconds() / 86400, 2),
            },
            "note_count": len(note_records),
            "pre_admission_note_count": int(ep_notes["pre_admission"].sum()),
            "notes": note_records,
        })

        index_rows.append({
            "episode_id": aid,
            "pseudonym": pseudo[pid],
            "age": int(p["age"]),
            "age_band": pd.cut([p["age"]], bins=AGE_BINS, right=False,
                               labels=AGE_LABELS)[0],
            "is_paediatric": bool(p["age"] < 18),
            "admission_method": adm["admission_method"],
            "department": adm["department"],
            "setting_tags": "|".join(adm["setting_tags"]),
            "note_count": len(note_records),
            "span_days": round((last - first).total_seconds() / 86400, 2),
            "total_chars": int(ep_notes["clean_note_text"].str.len().sum()),
        })

    # ---- write ------------------------------------------------------------
    (out_dir / "episodes.json").write_text(
        json.dumps(episodes, indent=2, ensure_ascii=False), encoding="utf-8")

    idx = pd.DataFrame(index_rows).sort_values("episode_id")
    idx.to_csv(out_dir / "episode_index.csv", index=False)

    # ---- report -----------------------------------------------------------
    print("PREPROCESSING COMPLETE")
    print(f"  raw rows            patients={raw_counts['patients']} "
          f"admissions={raw_counts['admissions']} notes={raw_counts['notes']}")
    print(f"  after deduplication patients={len(patients)} "
          f"episodes={len(episodes)} notes={len(notes)}")
    print(f"  paediatric episodes {int(idx['is_paediatric'].sum())} / {len(idx)}")
    print(f"  aged 65+            {int((idx['age'] >= 65).sum())}")
    print(f"  notes per episode   min={idx['note_count'].min()} "
          f"median={idx['note_count'].median()} max={idx['note_count'].max()}")
    print(f"  episode span (days) min={idx['span_days'].min()} "
          f"median={idx['span_days'].median()} max={idx['span_days'].max()}")
    print(f"  pre-admission notes {int(sum(e['pre_admission_note_count'] for e in episodes))}")
    print(f"  chars per episode   min={idx['total_chars'].min()} "
          f"median={int(idx['total_chars'].median())} max={idx['total_chars'].max()}")
    print(f"\n  wrote {out_dir/'episodes.json'}")
    print(f"  wrote {out_dir/'episode_index.csv'}")
    print(f"  wrote {out_dir/'pseudonyms.csv'}")
    return 0


if __name__ == "__main__":
    sys.exit(main())

#!/usr/bin/env python3
"""
generate_summaries.py — Stage 2 of the validation-study data pipeline.

Generates a clinician-facing discharge summary for each episode using a LOCAL
open-weights model served by Ollama. No data leaves the machine.

IMPORTANT — WHAT THIS IS AND IS NOT
  This script CALLS an existing, pre-trained, general-purpose language model.
  It does not build, train, fine-tune or evaluate a model. The model is treated
  as a black box that produces the artefacts which the study then validates.
  In the dissertation this belongs in Data Preparation, not Implementation.

STRUCTURE
  Summaries follow the NHS England / PRSB eDischarge Summary Standard v2.1
  (Open Government Licence v3.0). The 6 mandatory sections are always requested;
  selected 'required' sections are included because they carry the clinical
  content this study is about. Using a published standard means an omitted
  section is an objectively defined error, not one this project invented.

CITATIONS
  The model is asked to tag every factual statement with the numbered note(s)
  supporting it, e.g. "[N12]". These tags are NOT ground truth — model-generated
  citations are known to be unreliable, and that unreliability is itself part of
  what the validation interface exists to surface. They are a feature under test.

DETERMINISM
  temperature=0 and a fixed seed are set so a marker can reproduce the outputs.
  The model tag, all parameters and a SHA-256 of the prompt template are written
  to out/generation_manifest.json for the appendix.

USAGE
  python scripts/generate_summaries.py --dry-run --limit 2   # inspect prompts, no model
  python scripts/generate_summaries.py --model llama3.1:8b --limit 3
  python scripts/generate_summaries.py --model llama3.1:8b   # all 50

--------------------------------------------------------------------------
ATTRIBUTION: This file was written with AI assistance (Claude, Anthropic) and
reviewed/adapted by the author. Retained per City St George's MSc Project
Guidance, Appendix C items 7 and 9.
--------------------------------------------------------------------------
"""

from __future__ import annotations

import argparse
import hashlib
import json
import sys
import time
from datetime import datetime, timezone
from pathlib import Path

try:
    import requests
except ImportError:
    requests = None


OLLAMA_URL = "http://localhost:11434"

# --------------------------------------------------------------------------
# PRSB eDischarge Summary Standard v2.1 — section specification
# Mandatory sections are marked M; the required sections retained here are
# marked R. Source: https://theprsb.org/standards/edischargesummary/
# --------------------------------------------------------------------------
PRSB_SECTIONS = [
    ("Patient demographics",            "M", "Name, date of birth, NHS number, age, sex."),
    ("Admission details",               "R", "Date and time of admission, method of admission, ward/department."),
    ("Discharge details",               "R", "Date of discharge and discharge destination, if documented."),
    ("Diagnoses",                       "M", "Primary diagnosis first, then secondary diagnoses. Use the diagnosis actually recorded."),
    ("Clinical summary",                "M", "Narrative of the admission: presentation, key investigations, treatment, and progress."),
    ("Procedures",                      "R", "Any operations or procedures performed, with dates."),
    ("Investigations and results",      "R", "Key investigations and their results. Include pending results if documented."),
    ("Medications and medical devices", "R", "Discharge medications with dose, route and frequency. State changes made during admission."),
    ("Allergies and adverse reactions", "M", "State explicitly. If none are recorded, write 'None recorded'."),
    ("Plan and requested actions",      "R", "Follow-up arrangements, actions for the GP, and monitoring required."),
    ("Information provided to patient", "R", "What the patient or carer was told, if documented."),
    ("Person completing record",        "M", "Name and role of the clinician, as recorded in the notes."),
]

PROMPT_TEMPLATE = """You are a hospital doctor writing a discharge summary for the patient's GP.

You will be given the complete set of clinical notes from one hospital episode, \
in chronological order and numbered [N1], [N2], [N3] and so on.

Write a discharge summary using EXACTLY the section headings listed below, in \
this order. Use a level-2 markdown heading (##) for each section.

{section_spec}

RULES — follow all of them:
1. Use ONLY information present in the numbered notes. Do not add clinical \
knowledge of your own, and do not infer anything the notes do not state.
2. After every factual statement, cite the note or notes it came from, in \
square brackets: for example "Commenced on nimodipine 60mg four-hourly [N7]." \
A statement drawing on several notes cites all of them: "[N7][N12]".
3. If a section has no supporting information in the notes, write the heading \
followed by "Not documented." Do not omit the heading and do not invent content.
4. Write in the register of a UK hospital discharge summary: concise, factual, \
professional. Standard clinical abbreviations are acceptable.
5. Do not include any commentary, preamble or explanation outside the summary \
itself. Begin directly with the first heading.
6. Do NOT repeat the guidance text that follows each heading above. Write the \
heading, then the content. Do not use the guidance as a label.
7. The Clinical summary section must carry citations too. Every sentence in it \
needs at least one note reference.
8. Do not indent lines. Start each line at the left margin.

=== PATIENT RECORD ===
{demographics}

=== CLINICAL NOTES FOR THIS EPISODE ===
{notes_block}
=== END OF NOTES ===

Write the discharge summary now."""


def build_section_spec() -> str:
    """Render the section list.

    The guidance for each heading is placed after an arrow rather than on an
    indented line beneath it. Earlier runs showed the model echoing indented
    guidance back into the output as a label ("Name and role of the clinician,
    as recorded in the notes: Dr ..."), which is a formatting artefact of the
    prompt rather than anything about the clinical content.
    """
    lines = []
    for name, conformance, guidance in PRSB_SECTIONS:
        tag = "MANDATORY" if conformance == "M" else "required"
        lines.append(f"## {name}  ({tag})  -> {guidance}")
    return "\n".join(lines)


def build_notes_block(episode: dict) -> tuple[str, dict]:
    """Render the episode's notes as a numbered block.

    Simple integer labels (N1, N2, ...) are used rather than UUIDs because
    smaller local models cite short tokens far more reliably. The mapping back
    to real note_ids is returned and stored alongside the output.
    """
    parts, mapping = [], {}
    for note in episode["notes"]:
        label = f"N{note['seq']}"
        mapping[label] = note["note_id"]
        stamp = note["timestamp"].replace("T", " ")
        flag = "  (PRE-ADMISSION)" if note["pre_admission"] else ""
        parts.append(
            f"[{label}] {stamp} | {note['note_type']} | {note['note_subject']}{flag}\n"
            f"{note['text'].strip()}\n"
        )
    return "\n".join(parts), mapping


def build_demographics(episode: dict) -> str:
    """Structured patient and admission details.

    These come from the patient and admission tables, not the notes. Without
    them the model reports PRSB mandatory demographic fields as "Not documented"
    — an omission caused by what we withheld rather than by the model, which
    would contaminate the error key. A real deployment reads these straight
    from the patient record, so supplying them is the faithful setup.
    """
    p, a = episode["patient"], episode["admission"]
    return (
        f"Name: {p['full_name']}\n"
        f"Date of birth: {p['date_of_birth']}\n"
        f"NHS number: {p['nhs_number']}\n"
        f"Age: {p['age']}\n"
        f"Sex: {p['gender_identity']}\n"
        f"Admission date/time: {a['timestamp'].replace('T', ' ')}\n"
        f"Method of admission: {a['method']}\n"
        f"Ward/department: {a['department']}"
    )


def build_prompt(episode: dict) -> tuple[str, dict]:
    notes_block, mapping = build_notes_block(episode)
    prompt = PROMPT_TEMPLATE.format(
        section_spec=build_section_spec(),
        demographics=build_demographics(episode),
        notes_block=notes_block,
    )
    return prompt, mapping


def call_ollama(model: str, prompt: str, num_ctx: int, seed: int, timeout: int,
                no_think: bool = True) -> dict:
    if requests is None:
        raise RuntimeError("The 'requests' package is required: pip install requests")
    payload = {
        "model": model,
        "prompt": prompt,
        "stream": False,
        # Keep the model resident between episodes. Ollama's default unloads it
        # after 5 minutes idle, which on CPU costs ~100s to reload each time —
        # over an hour wasted across a 50-episode run.
        "keep_alive": "60m",
        "options": {
            # Ollama defaults num_ctx to 2048, which would SILENTLY TRUNCATE these
            # prompts (median ~5k tokens, up to ~9.5k). This must be set explicitly.
            "num_ctx": num_ctx,
            "temperature": 0.0,
            "top_p": 1.0,
            "seed": seed,
        },
    }
    # Reasoning ("thinking") models such as the Qwen3 family emit a long internal
    # monologue before answering, and Ollama returns it in a separate `thinking`
    # field. On CPU that monologue can consume the entire generation budget, so
    # the summary never gets written. Ask the model to skip it.
    if no_think:
        payload["think"] = False

    response = requests.post(f"{OLLAMA_URL}/api/generate", json=payload, timeout=timeout)

    # Models that do not support the `think` flag reject the request; retry plainly.
    if response.status_code >= 400 and no_think:
        body = response.text.lower()
        if "think" in body:
            payload.pop("think", None)
            response = requests.post(f"{OLLAMA_URL}/api/generate",
                                     json=payload, timeout=timeout)

    if response.status_code != 200:
        # Ollama explains the real problem in the response body (out of memory,
        # context too large, model failed to load). Surface it instead of the
        # bare HTTP status, which tells you nothing useful.
        try:
            detail = response.json().get("error", response.text)
        except Exception:
            detail = response.text
        raise RuntimeError(f"Ollama returned {response.status_code}: {detail[:600]}")
    return response.json()


def check_ollama(model: str) -> bool:
    if requests is None:
        print("ERROR: pip install requests", file=sys.stderr)
        return False
    try:
        tags = requests.get(f"{OLLAMA_URL}/api/tags", timeout=10).json()
    except Exception as exc:
        print(f"ERROR: cannot reach Ollama at {OLLAMA_URL} ({exc})", file=sys.stderr)
        print("       Start it with:  ollama serve", file=sys.stderr)
        return False

    available = [m["name"] for m in tags.get("models", [])]
    print(f"Ollama reachable. Models installed: {available or '(none)'}")
    if not any(m == model or m.startswith(model.split(':')[0]) for m in available):
        print(f"\nERROR: model '{model}' is not installed.", file=sys.stderr)
        print(f"       Pull it with:  ollama pull {model}", file=sys.stderr)
        return False
    return True


def main() -> int:
    ap = argparse.ArgumentParser(description=__doc__,
                                 formatter_class=argparse.RawDescriptionHelpFormatter)
    ap.add_argument("--episodes", default="out/episodes.json")
    ap.add_argument("--out", default="out/summaries_clinician")
    ap.add_argument("--model", default="mistral:latest")
    ap.add_argument("--num-ctx", type=int, default=16384,
                    help="context window; must exceed the longest prompt (~9.5k tokens)")
    ap.add_argument("--seed", type=int, default=20260821)
    ap.add_argument("--timeout", type=int, default=3600, help="per-episode timeout, seconds")
    ap.add_argument("--limit", type=int, default=None, help="process only the first N episodes")
    ap.add_argument("--dry-run", action="store_true",
                    help="write prompts only; do not call the model")
    ap.add_argument("--overwrite", action="store_true",
                    help="regenerate summaries that already exist")
    ap.add_argument("--smoke-test", action="store_true",
                    help="send one tiny prompt to check the model loads at all, then stop")
    args = ap.parse_args()

    if args.smoke_test:
        if not check_ollama(args.model):
            return 1
        print(f"\nSmoke test: loading {args.model} at num_ctx={args.num_ctx} ...")
        t0 = time.time()
        try:
            out = call_ollama(args.model, "Reply with the single word: ready",
                              args.num_ctx, args.seed, 600)
        except Exception as exc:
            print(f"\nFAILED after {time.time()-t0:.0f}s\n  {exc}", file=sys.stderr)
            print("\nMost likely: not enough free memory for this model at this "
                  "context size.\n  Try:  --num-ctx 4096\n  Or a smaller model: "
                  "--model mistral:latest", file=sys.stderr)
            return 1
        print(f"OK in {time.time()-t0:.0f}s. Model replied: "
              f"{out.get('response','').strip()[:120]!r}")
        print("The model loads fine at this context size.")
        return 0

    episodes = json.loads(Path(args.episodes).read_text(encoding="utf-8"))
    if args.limit:
        episodes = episodes[:args.limit]

    out_dir = Path(args.out)
    (out_dir / "prompts").mkdir(parents=True, exist_ok=True)

    if not args.dry_run and not check_ollama(args.model):
        return 1

    prompt_hash = hashlib.sha256(PROMPT_TEMPLATE.encode()).hexdigest()[:16]
    records, started = [], time.time()

    for i, episode in enumerate(episodes, 1):
        eid = episode["episode_id"]
        target = out_dir / f"{eid}.md"
        if target.exists() and not args.overwrite and not args.dry_run:
            print(f"[{i}/{len(episodes)}] {eid[:8]} — exists, skipping")
            continue

        prompt, mapping = build_prompt(episode)
        (out_dir / "prompts" / f"{eid}.txt").write_text(prompt, encoding="utf-8")
        approx_tokens = len(prompt) // 4

        if approx_tokens > args.num_ctx * 0.9:
            print(f"  WARNING: prompt ~{approx_tokens} tokens is close to "
                  f"num_ctx={args.num_ctx}; raise --num-ctx", file=sys.stderr)

        if args.dry_run:
            print(f"[{i}/{len(episodes)}] {eid[:8]} — prompt written "
                  f"(~{approx_tokens} tokens, {episode['note_count']} notes)")
            records.append({"episode_id": eid, "approx_prompt_tokens": approx_tokens,
                            "note_count": episode["note_count"], "dry_run": True})
            continue

        t0 = time.time()
        try:
            result = call_ollama(args.model, prompt, args.num_ctx, args.seed, args.timeout)
        except Exception as exc:
            print(f"[{i}/{len(episodes)}] {eid[:8]} — FAILED: {exc}", file=sys.stderr)
            records.append({"episode_id": eid, "error": str(exc)})
            continue

        elapsed = time.time() - t0

        # A reasoning model may put everything in `thinking` and leave `response`
        # empty. Save whatever we actually got rather than writing a 0-byte file,
        # and say plainly which field it came from.
        text = (result.get("response") or "").strip()
        thinking = (result.get("thinking") or "").strip()
        source = "response"
        if not text and thinking:
            text = thinking
            source = "thinking (model reasoned but never produced a final answer)"

        if not text:
            print(f"[{i}/{len(episodes)}] {eid[:8]} — EMPTY OUTPUT after "
                  f"{elapsed:.0f}s. The model generated "
                  f"{result.get('eval_count', '?')} tokens but returned nothing "
                  f"usable. Try a non-reasoning model: --model mistral:latest",
                  file=sys.stderr)
            records.append({"episode_id": eid, "error": "empty output",
                            "seconds": round(elapsed, 1),
                            "eval_count": result.get("eval_count")})
            continue

        target.write_text(text, encoding="utf-8")
        (out_dir / f"{eid}.notemap.json").write_text(
            json.dumps(mapping, indent=2), encoding="utf-8")

        records.append({
            "episode_id": eid,
            "note_count": episode["note_count"],
            "approx_prompt_tokens": approx_tokens,
            "prompt_eval_count": result.get("prompt_eval_count"),
            "eval_count": result.get("eval_count"),
            "seconds": round(elapsed, 1),
            "output_chars": len(text),
            "output_source": source,
        })
        print(f"[{i}/{len(episodes)}] {eid[:8]} — {elapsed:.0f}s, "
              f"{result.get('eval_count', '?')} tokens, {len(text)} chars written"
              + ("" if source == "response" else f"  [from {source}]"))

    # Merge with any previous manifest rather than replacing it. A later partial
    # run (e.g. regenerating two failed episodes) would otherwise discard the
    # timings and token counts from the original full run, which are needed for
    # the methodology chapter.
    manifest_path = out_dir / "generation_manifest.json"
    previous = []
    if manifest_path.exists():
        try:
            old = json.loads(manifest_path.read_text(encoding="utf-8"))
            done = {r.get("episode_id") for r in records}
            previous = [r for r in old.get("episodes", [])
                        if r.get("episode_id") not in done]
        except Exception:
            pass
    records = previous + records

    manifest = {
        "generated_at": datetime.now(timezone.utc).isoformat(),
        "model": args.model,
        "backend": "ollama (local, no network egress)",
        "parameters": {"num_ctx": args.num_ctx, "temperature": 0.0,
                       "top_p": 1.0, "seed": args.seed},
        "prompt_template_sha256_16": prompt_hash,
        "prsb_standard": "NHS England / PRSB eDischarge Summary Standard v2.1 (OGL v3.0)",
        "summary_type": "clinician-facing",
        "dry_run": args.dry_run,
        "episodes": records,
    }
    manifest_path.write_text(json.dumps(manifest, indent=2), encoding="utf-8")

    ok = [r for r in records if "error" not in r and not r.get("dry_run")]
    print(f"\nDone in {time.time() - started:.0f}s. "
          f"{len(ok)} generated, {len([r for r in records if 'error' in r])} failed.")
    if ok:
        times = [r["seconds"] for r in ok]
        print(f"  per-episode: min {min(times):.0f}s  "
              f"median {sorted(times)[len(times)//2]:.0f}s  max {max(times):.0f}s")
    print(f"  manifest: {out_dir/'generation_manifest.json'}")
    return 0


if __name__ == "__main__":
    sys.exit(main())

#!/usr/bin/env python3
"""
check_summaries.py — quality check across the generated summary set.

Reports, for all 50 summaries at once:
  * whether every PRSB section heading is present
  * how many note citations each summary carries, and where they fall
  * whether the prompt-guidance text is being echoed back as a label
  * whether mandatory demographic fields were populated
  * length distribution

Run:  python scripts\\check_summaries.py

--------------------------------------------------------------------------
ATTRIBUTION: Written with AI assistance (Claude, Anthropic), reviewed and
adapted by the author. Retained per MSc Project Guidance Appendix C, items 7/9.
--------------------------------------------------------------------------
"""

from __future__ import annotations

import json
import re
import statistics
from collections import Counter
from pathlib import Path

SUMMARY_DIR = Path("out/summaries_clinician")

PRSB_HEADINGS = [
    "Patient demographics", "Admission details", "Discharge details", "Diagnoses",
    "Clinical summary", "Procedures", "Investigations and results",
    "Medications and medical devices", "Allergies and adverse reactions",
    "Plan and requested actions", "Information provided to patient",
    "Person completing record",
]

# Fragments of the prompt guidance. If these appear in the OUTPUT the model is
# echoing my instructions back as labels rather than writing content.
ECHO_MARKERS = [
    "Name and role of the clinician, as recorded in the notes",
    "Discharge medications with dose, route and frequency",
    "Date and time of admission, method of admission",
    "Primary diagnosis first, then secondary diagnoses",
    "Narrative of the admission: presentation",
    "Follow-up arrangements, actions for the GP",
    "What the patient or carer was told",
    "(MANDATORY)", "(required)", "->",
]

CITE = re.compile(r"\[N\d+\]")


def section_body(text: str, heading: str) -> str:
    """Return the text under one heading, up to the next heading."""
    m = re.search(rf"^##\s*{re.escape(heading)}\s*$", text, re.M | re.I)
    if not m:
        return ""
    rest = text[m.end():]
    nxt = re.search(r"^##\s", rest, re.M)
    return rest[:nxt.start()] if nxt else rest


def main() -> None:
    files = sorted(SUMMARY_DIR.glob("*.md"))
    if not files:
        print(f"No .md files found in {SUMMARY_DIR.resolve()}")
        return

    print(f"Checking {len(files)} summaries in {SUMMARY_DIR}\n")

    lengths, cite_counts, missing_headings = [], [], Counter()
    echo_hits, no_dob, empty_clinical, clin_no_cites = [], [], [], []
    cites_by_section = Counter()
    not_documented = Counter()

    for f in files:
        text = f.read_text(encoding="utf-8", errors="replace")
        sid = f.stem[:8]
        lengths.append(len(text))
        cite_counts.append(len(CITE.findall(text)))

        for h in PRSB_HEADINGS:
            body = section_body(text, h)
            if not re.search(rf"^##\s*{re.escape(h)}\s*$", text, re.M | re.I):
                missing_headings[h] += 1
            else:
                cites_by_section[h] += len(CITE.findall(body))
                if re.search(r"not documented", body, re.I):
                    not_documented[h] += 1

        for marker in ECHO_MARKERS:
            if marker in text:
                echo_hits.append((sid, marker))
                break

        demo = section_body(text, "Patient demographics")
        if re.search(r"date of birth[^\n]*not documented", demo, re.I) or \
           not re.search(r"date of birth", demo, re.I):
            no_dob.append(sid)

        clin = section_body(text, "Clinical summary").strip()
        if len(clin) < 100:
            empty_clinical.append(sid)
        elif not CITE.search(clin):
            clin_no_cites.append(sid)

    print("=== STRUCTURE ===")
    if missing_headings:
        for h, n in missing_headings.most_common():
            print(f"  MISSING in {n:2d}/{len(files)}:  {h}")
    else:
        print(f"  All 12 PRSB headings present in all {len(files)} summaries.")

    print("\n=== LENGTH (characters) ===")
    print(f"  min {min(lengths)}  median {int(statistics.median(lengths))}  "
          f"mean {int(statistics.mean(lengths))}  max {max(lengths)}")

    print("\n=== CITATIONS ===")
    print(f"  per summary: min {min(cite_counts)}  "
          f"median {int(statistics.median(cite_counts))}  max {max(cite_counts)}")
    print(f"  summaries with zero citations: {sum(1 for c in cite_counts if c == 0)}")
    print("  total citations by section:")
    for h in PRSB_HEADINGS:
        print(f"    {cites_by_section[h]:5d}  {h}")

    print("\n=== PROMPT FIX CHECKS ===")
    print(f"  echoing guidance text as labels: {len(echo_hits)}/{len(files)}"
          + (f"   e.g. {echo_hits[0]}" if echo_hits else "   (fixed)"))
    print(f"  date of birth missing:           {len(no_dob)}/{len(files)}"
          + (f"   {no_dob[:6]}" if no_dob else "   (fixed)"))
    print(f"  clinical summary empty/short:    {len(empty_clinical)}/{len(files)}"
          + (f"   {empty_clinical[:6]}" if empty_clinical else ""))
    print(f"  clinical summary lacks citations:{len(clin_no_cites)}/{len(files)}"
          + (f"   {clin_no_cites[:6]}" if clin_no_cites else "   (fixed)"))

    print("\n=== 'Not documented' BY SECTION ===")
    for h, n in not_documented.most_common():
        print(f"  {n:3d}/{len(files)}  {h}")

    man = SUMMARY_DIR / "generation_manifest.json"
    if man.exists():
        m = json.loads(man.read_text(encoding="utf-8"))
        eps = [e for e in m.get("episodes", []) if "seconds" in e]
        print("\n=== RUN ===")
        print(f"  model: {m.get('model')}   params: {m.get('parameters')}")
        print(f"  prompt sha256(16): {m.get('prompt_template_sha256_16')}")
        if eps:
            secs = [e["seconds"] for e in eps]
            print(f"  episodes timed: {len(eps)}   "
                  f"per-episode seconds: min {min(secs):.0f} "
                  f"median {statistics.median(secs):.0f} max {max(secs):.0f}")
            print(f"  total generation time: {sum(secs)/3600:.1f} hours")


if __name__ == "__main__":
    main()

def norm(t):
    t = t.lower()
    t = re.sub(r"([a-z])([A-Z])", r"\1 \2", t)
    t = t.replace("&", " and ")
    t = re.sub(r"[^a-z0-9\s/-]", " ", t)
    return t

def toks(t):
    out = []
    for w in norm(t).split():
        w = w.strip("-/")
        if len(w) < 3 or w in STOP or w.isdigit():
            continue
        out.append(w)
    return out

def run(records, df_max, min_hits):
    n = len(records)
    df = collections.Counter()
    for txt in records.values():
        df.update(set(toks(txt)))
    distinctive = {w for w, c in df.items() if c / n <= df_max}

    predicted, reasons = {}, {}
    for k, s in STATEMENTS.items():
        if k in FIELD_CHECKS:
            predicted[k] = []
            continue
        keys = set(toks(s)) & distinctive
        hits = {}
        for rid, txt in records.items():
            shared = keys & set(toks(txt))
            if len(shared) >= min_hits:
                hits[rid] = sorted(shared)
        predicted[k] = sorted(hits, key=lambda r: int(r[1:]))
        reasons[k] = hits
    return predicted, reasons, distinctive

def score(pred):
    tp = fp = fn = 0
    per = {}
    for k, truth in TRUTH.items():
        if k in FIELD_CHECKS:
            continue
        p, t = set(pred[k]), set(truth)
        a, b, c = len(p & t), len(p - t), len(t - p)
        tp += a; fp += b; fn += c
        per[k] = (a, b, c)
    prec = tp / (tp + fp) if tp + fp else 0
    rec = tp / (tp + fn) if tp + fn else 0
    f1 = 2 * prec * rec / (prec + rec) if prec + rec else 0
    return prec, rec, f1, per, (tp, fp, fn)

    print(f"links: {tp} correct, {fp} spurious, {fn} missed  |  "
          f"precision {p:.2f}  recall {r:.2f}  F1 {f1:.2f}\n")

    print(f"{'statement':10} {'hit':>4} {'extra':>6} {'miss':>5}   what went wrong")
    for k in STATEMENTS:
        if k in FIELD_CHECKS:
            print(f"{k:10} {'—':>4} {'—':>6} {'—':>5}   settled against patients.csv")
            continue
        a, b, c = per[k]
        extra = sorted(set(pred[k]) - set(TRUTH[k]), key=lambda x: int(x[1:]))
        miss = sorted(set(TRUTH[k]) - set(pred[k]), key=lambda x: int(x[1:]))
        note = ""
        if extra: note += "spurious " + ",".join(extra) + " "
        if miss:  note += "missed " + ",".join(miss)
        print(f"{k:10} {a:>4} {b:>6} {c:>5}   {note or 'exact match'}")

    print("\nWHY EACH LINK WAS DRAWN (the reason the interface would show)")
    for k in ("trip", "nimo", "bloods", "dest"):
        print(f"\n  {k} — \"{STATEMENTS[k][:64]}...\"")
        if not reasons.get(k):
            print("     no record shares a distinctive term  ->  no evidence")
        for rid, shared in sorted(reasons[k].items(), key=lambda x: int(x[0][1:])):
            mark = "ok " if rid in TRUTH[k] else "XX "
            print(f"     {mark}{rid:4} shares: {', '.join(shared[:6])}")

if __name__ == "__main__":
    main()

        hits = {}
        for rid, txt in records.items():
            if method == "A":
                shared = st & distinctive & set(toks(txt))
                if len(shared) >= min_hits:
                    hits[rid] = (len(shared), sorted(shared), "")
            elif method == "B":
                shared = st & set(toks(txt))
                sc = sum(idf.get(w, 0) for w in shared)
                if sc >= thr:
                    hits[rid] = (sc, sorted(shared, key=lambda w: -idf.get(w, 0)), "")
            else:                                     # C — best single line
                best = (0, [], "")
                for ln in lines(txt):
                    shared = st & set(toks(ln))
                    sc = sum(idf.get(w, 0) for w in shared)
                    if sc > best[0]:
                        best = (sc, sorted(shared, key=lambda w: -idf.get(w, 0)), ln.strip()[:90])
                if best[0] >= thr:
                    hits[rid] = best
        pred[k] = sorted(hits, key=lambda r: int(r[1:]))
        why[k] = hits
    return pred, why

def score(pred):
    tp = fp = fn = 0
    per = {}
    for k, t in TRUTH.items():
        if k in FIELD: continue
        p, t = set(pred[k]), set(t)
        a, b, c = len(p & t), len(p - t), len(t - p)
        tp += a; fp += b; fn += c
        per[k] = (a, b, c)
    pr = tp / (tp + fp) if tp + fp else 0
    rc = tp / (tp + fn) if tp + fn else 0
    return pr, rc, (2*pr*rc/(pr+rc) if pr+rc else 0), per, (tp, fp, fn)

def main():
    ep = [e for e in json.load(open("out/episodes.json"))
          if e["episode_id"].startswith("0219fb71")][0]
    recs = {f"N{n['seq']}": n["text"] for n in ep["notes"]}
    print(f"episode 0219fb71 · {len(recs)} records · 14 statements scored "
          f"(2 settled as field checks)\n")

    print(f"  {'method':34} {'precision':>10} {'recall':>8} {'F1':>7}")
    best = {}
    for label, m, params in [("A  distinctive-term overlap", "A", [(0, 0.30, 2)]),
                             ("B  idf-weighted, whole record", "B", [(t, 0, 0) for t in (2,3,4,5,6,7,8)]),
                             ("C  idf-weighted, best line", "C", [(t, 0, 0) for t in (2,3,4,5,6,7,8)])]:
        top = None
        for thr, dmx, mh in params:
            pred, why = score_all(recs, m, thr, dmx or 0.30, mh or 2)
            pr, rc, f1, per, tot = score(pred)
            if top is None or f1 > top[0]:
                top = (f1, pr, rc, thr, pred, why, per, tot)
        f1, pr, rc, thr, pred, why, per, tot = top
        tag = f"{label}" + (f"  (threshold {thr})" if m != "A" else "")
        print(f"  {tag:34} {pr:>10.2f} {rc:>8.2f} {f1:>7.2f}")
        best[m] = top

    bm = max(best, key=lambda m: best[m][0])          # by F1, not hard-coded
    f1, pr, rc, thr, pred, why, per, (tp, fp, fn) = best[bm]
    print(f"\nBEST METHOD BY F1: {bm} — F1 {f1:.2f} — "
          f"{tp} correct, {fp} spurious, {fn} missed\n")
    print(f"{'statement':10} {'hit':>4} {'extra':>6} {'miss':>5}   remaining error")
    for k in STATEMENTS:
        if k in FIELD:
            print(f"{k:10} {'—':>4} {'—':>6} {'—':>5}   field check"); continue
        a, b, c = per[k]
        ex = sorted(set(pred[k]) - set(TRUTH[k]), key=lambda x: int(x[1:]))
        ms = sorted(set(TRUTH[k]) - set(pred[k]), key=lambda x: int(x[1:]))
        msg = ("spurious " + ",".join(ex) + " " if ex else "") + ("missed " + ",".join(ms) if ms else "")
        print(f"{k:10} {a:>4} {b:>6} {c:>5}   {msg or 'exact'}")

    print("\nTHE TWO RECORDS THAT MATTER MOST — does the method find them?")
    for k, want in (("trip", ["N26","N29"]),):
        for rid in want:

            det = why[k].get(rid)
            print(f"  {rid}  {'FOUND' if got else 'MISSED'}"

"""
Linking summary statements to records: three methods compared on episode 0219fb71,
scored against hand-verified links.

A  distinctive-term overlap    tokens rarer than a cutoff; N shared terms required
B  idf-weighted, record level  every content token scores log(N/df); threshold on the sum
C  idf-weighted, sentence level  as B, but the record is scored by its best single line

WHAT CHANGED FROM link_experiment2.py, AND WHY
----------------------------------------------
v2 chose the best threshold within each method by F1, and then the best method by F1.
F1 weights precision and recall equally. For this task they are not equal.

The linking step does not decide anything. It puts candidate records in front of a
clinician, who decides. So:

  a MISSED link  -> the clinician never sees a record that contradicts the summary.
                    This is the dangerous error.
  a SPURIOUS link -> the clinician reads a record that turns out to be irrelevant.
                    This costs seconds and is self-correcting.

Optimising F1 treats those as equally bad and therefore picks the wrong operating
point for a screen with a human filter downstream. This version changes nothing about
the methods — it reports the whole precision/recall trade-off and three operating
points side by side, so the choice is visible rather than baked in:

  BEST F1       reproduces the v2 headline, so the earlier figure stays checkable
  BEST F2       F-beta with beta=2, which weights recall twice as heavily as precision
  MAX RECALL    the most complete retrieval available, whatever precision costs

It also prints the concrete cost of each point: how many irrelevant records a
clinician would have to read, per statement, to buy that recall.

Nothing here proves that a high-recall operating point is usable. That is a question
for the evaluation, not for this script.
"""

def norm(t):
    t = re.sub(r"([a-z])([A-Z])", r"\1 \2", t).lower().replace("&"," and ")
    return re.sub(r"[^a-z0-9\s/-]", " ", t)

def toks(t):
    return [w.strip("-/") for w in norm(t).split()
            if len(w.strip("-/")) >= 3 and w.strip("-/") not in STOP and not w.isdigit()]

def lines(t):
    parts = re.split(r"(?<=[.;])\s+|\n+|(?<=\d\.)\s", t)
    return [p for p in parts if len(p.strip()) > 12]

def score_all(records, method, thr, df_max=0.30, min_hits=2):
    N = len(records)
    df = collections.Counter()
    for t in records.values():
        df.update(set(toks(t)))
    idf = {w: math.log(N / c) for w, c in df.items()}
    distinctive = {w for w, c in df.items() if c / N <= df_max}

    pred, why = {}, {}
    for k, s in STATEMENTS.items():

            pred[k] = []; continue

        hits = {}

            elif method == "B":

                sc = sum(idf.get(w, 0) for w in shared)
                if sc >= thr:

            else:                                     # C — best single line
                best = (0, [], "")
                for ln in lines(txt):
                    shared = st & set(toks(ln))
                    sc = sum(idf.get(w, 0) for w in shared)
                    if sc > best[0]:

                if best[0] >= thr:

        pred[k] = sorted(hits, key=lambda r: int(r[1:]))

def fbeta(pr, rc, beta):
    b2 = beta * beta
    return (1 + b2) * pr * rc / (b2 * pr + rc) if (b2 * pr + rc) else 0

def score(pred):
    tp = fp = fn = 0
    per = {}
    for k, t in TRUTH.items():
        if k in FIELD: continue
        p, t = set(pred[k]), set(t)
        a, b, c = len(p & t), len(p - t), len(t - p)
        tp += a; fp += b; fn += c
        per[k] = (a, b, c)
    pr = tp / (tp + fp) if tp + fp else 0
    rc = tp / (tp + fn) if tp + fn else 0
    return dict(pr=pr, rc=rc, f1=fbeta(pr, rc, 1), f2=fbeta(pr, rc, 2),
                per=per, tp=tp, fp=fp, fn=fn)

def sweep(recs):
    """Every method at every threshold. One row per operating point."""
    # v2 swept thresholds 2..8 only. The recall end of the curve lives below 2, and that
    # is the half of the trade-off this script exists to show, so the sweep starts lower.
    THR = [0.5, 1, 1.5, 2, 3, 4, 5, 6, 7, 8]
    grid = [("A", "distinctive-term overlap", [(0, 0.30, 2)])]
    grid += [("B", "idf-weighted, whole record", [(t, 0, 0) for t in THR])]
    grid += [("C", "idf-weighted, best line",  [(t, 0, 0) for t in THR])]
    rows = []
    for m, label, params in grid:
        for thr, dmx, mh in params:
            pred, why = score_all(recs, m, thr, dmx or 0.30, mh or 2)
            s = score(pred)
            s.update(method=m, label=label, thr=thr, pred=pred, why=why)
            rows.append(s)
    return rows

/* Setting a verdict also selects the line, so the evidence and the source row for the
   thing being judged are on screen at the moment of judging. Pressing the same mark
   again clears the verdict. */
function setVerdict(id,v){
  if(verdict.get(id)===v){ verdict.delete(id); reasons.delete(id); cited.delete(id); }
  else {
    verdict.set(id,v);
    if(v==="ok"){ reasons.delete(id); cited.delete(id); }
    /* start from whatever record is open, else the last one that disagrees or carries
       more — the latest is usually the one that matters. The reviewer can change it. */
    else if(!cited.has(id)){
      const s2=ST[id], last=a=>a.length?a[a.length-1][0]:null;
      cited.set(id, (id===sel&&selRec) || last(s2.ev.filter(e=>e[1]==="x"))
        || last(s2.ev.filter(e=>e[1]==="m")) || last(s2.ev) || null);
    }
  }
  sel=id; selGroup=null; render();
  if(verdict.get(id)==="flag"){
    const inp=document.getElementById("rin");
    if(inp){ inp.focus(); }
  }
}

/* Everything in this sentence is counted off the chart, so a deployed system could write
   it for any drug in any episode. The paragraph below it is the researcher's reading of
   this one case and is labelled as such — see §4.7. */
function autoDose(d){
  const tot=d.pts.length, car=d.pts.filter(p=>p[4]), n=car.length;
  const dt=r=>`${REC[r][0].slice(8,10)} Jan ${REC[r][0].slice(11,16)}`;
  const day=r=>`${REC[r][0].slice(8,10)} Jan`;
  if(!n) return `<b>${tot}</b> record${tot===1?"":"s"} name this drug and the summary reports `+
    `none of them.`;
  const lastCar=car[car.length-1][0], after=d.pts.slice(d.pts.indexOf(car[car.length-1])+1);
  const span=n===1?`on ${day(car[0][0])}`:
    day(car[0][0])!==day(lastCar)?`from ${day(car[0][0])} to ${day(lastCar)}`:
    n===2?`both on ${day(car[0][0])}`:`all on ${day(car[0][0])}`;
  if(!after.length) return `<b>${tot}</b> record${tot===1?"":"s"} name this drug and the summary `+
    `reports ${n===tot?"all "+n:n}, ${span}. Nothing in the record follows them.`;
  const stop=after.some(p=>p[1]===0);
  return `<b>${tot}</b> records name this drug. The summary reports <b>${n}</b>, ${span}. `+
    `<b>${after.length}</b> later record${after.length===1?"":"s"} follow, the last on `+
    `${dt(after[after.length-1][0])}`+
    (stop?`, including an instruction to stop the drug`:``)+
    `. None of them is in the summary.`;
}

/* One record, whole. The source row is printed on the card itself rather than in a panel
   underneath, so the reviewer reads the evidence and its provenance in one place and the
   words they are judging are never a paraphrase. R11. */
/* SummVis (Vig et al., 2021) draws coloured underlines on the source showing which of its
   phrases a summary reused; whatever carries no underline was not used. The same picture
   here, in reverse: the parts no statement carries are marked, so a reader sees coverage
   inside the record instead of having to hold the summary in memory while reading it. */
function markUncarried(r,text){
  /* the card now carries the note's own line structure, so a quote may span a
     line break — match on whitespace runs rather than on an exact substring */
  let out=text, n=0;
  SPANS.filter(x=>x[1]===r).forEach(([sid,,quote])=>{
    const q=quote.replace(/\.$/,"").trim();
    const rx=new RegExp(q.split(/\s+/)
      .map(w=>w.replace(/[.*+?^${}()|[\]\\]/g,"\\$&")).join("\\s+"));
    const m=out.match(rx);
    if(m && out.indexOf("<mark>"+m[0]+"</mark>")<0){
      out=out.replace(rx,`<u class="nc">${m[0]}</u>`); n++; }
  });
  return {html:out,n};
}
/* Before the rebuild, each card held a hand-cut excerpt — the clause the summary took and a
   few words either side. That made the clause easy to find and made the card a claim the
   reviewer could not check: the thing they were asked to trust was an abridgement made by the
   person asking them to trust it. The card now carries the note as written, typing errors and
   all. The cost is length. So the box opens at the highlighted clause rather than at the top,
   and its edges say where there is more. */
function frameCards(){
  document.querySelectorAll(".evl .rowtx").forEach(box=>{
    if(box.scrollHeight<=box.clientHeight+2) return;
    box.classList.add("more");
    const edges=()=>{
      box.style.setProperty("--fadeT",box.scrollTop>4?"22px":"0px");
      box.style.setProperty("--fadeB",
        box.scrollTop+box.clientHeight<box.scrollHeight-4?"28px":"0px");
    };
    const m=box.querySelector("mark")||box.querySelector("u.nc");
    if(m){
      const rb=box.getBoundingClientRect(), rm=m.getBoundingClientRect();
      box.scrollTop+=(rm.top-rb.top)-(box.clientHeight-rm.height)/2;
    }
    edges();
    box.addEventListener("scroll",edges,{passive:true});
  });
}
function card(r,st,extra,after){
  const [t,ty,row,text]=REC[r];
  const m=markUncarried(r,text);
  return `<li><button data-r="${r}" class="${st==="c"?"":st}${selRec===r?" on":""}">`+
    `<span class="rid">${extra||""}<b>${r}</b> · ${t.slice(8,10)}/${t.slice(5,7)}/2026 `+
    `${t.slice(11,16)} · ${svcTitle(r)}`+
    `${st?` · <span class="wd">${WORD[st]}</span>`:``}</span>`+
    `<span class="rowtx">${m.html}</span>`+
    (m.n?`<span class="ncnote">${m.n} line${m.n===1?"":"s"} here `+
      `${m.n===1?"is":"are"} carried by no statement in the summary — underlined above, and `+
      `listed at the foot of the screen</span>`:``)+
    `<span class="prov2">synthetic_clinical_notes.csv · row ${row}</span></button>`+
    (after||``)+`</li>`;
}

#!/usr/bin/env python3
"""
Rebuild the record cards in Validation_screen.html from the source notes.

WHY THIS EXISTS
The screen's REC table held hand-abridged excerpts averaging 153 characters
against source notes averaging 917 — 17% of the record. Twenty-five of the
thirty were not clean substrings: fragments had been stitched together,
typos silently corrected, and in nine places text appeared on a card that
is nowhere in the note it claims to quote. The screen's own help panel says
"Nothing is paraphrased", and the report says the cards carry "the
untruncated source text". Both were false.

WHAT IT DOES
  1. Replaces every REC text with the full note from episodes.json,
     line structure intact.
  2. Re-anchors each <mark> (the clause the summary took) in the full text,
     matching across line breaks.
  3. Reports any mark or span that cannot be anchored, rather than guessing.

WHAT IT DELIBERATELY DOES NOT DO
  Correct typos. The source says "Expresed", "sewek help", "managemnt",
  "imodipine". They stay. The project makes a finding out of "imodipine" —
  silently fixing the others while keeping that one would be incoherent,
  and the whole claim of the screen is that the card is the record.

    python rebuild_records.py  [--write]
"""
import json, re, sys, html as H, difflib
from pathlib import Path

HERE = Path(__file__).resolve().parent
SCREEN = HERE / "designs" / "Validation_screen.html"
EPISODES = HERE / "pipeline" / "out" / "episodes.json"
EPISODE = "0219fb71"
WRITE = "--write" in sys.argv

src = SCREEN.read_text(encoding="utf-8")
ep = [e for e in json.loads(EPISODES.read_text(encoding="utf-8"))
      if e["episode_id"].startswith(EPISODE)][0]
FULL = {f"N{n['seq']}": n for n in ep["notes"]}

# ---- current REC, for its metadata and its <mark> spans ----
OLD = {}
for m in re.finditer(r'^ (N\d+):\["([^"]*)","([^"]*)",(\d+),"(.*?)"\](?:,|\};)\s*$', src, re.M):
    rid, ts, typ, row, raw = m.groups()
    OLD[rid] = dict(ts=ts, typ=typ, row=int(row), raw=raw,
                    marks=[H.unescape(x) for x in re.findall(r"<mark>(.*?)</mark>", raw)])
assert len(OLD) == 30, f"expected 30 records, parsed {len(OLD)}"

def tidy(t):
    """Keep the note's line structure; drop trailing spaces and runs of blanks."""
    lines = [ln.rstrip() for ln in t.replace("\r\n", "\n").split("\n")]
    out, blank = [], 0
    for ln in lines:
        if ln.strip() == "":
            blank += 1
            if blank > 1: continue
        else:
            blank = 0
        out.append(ln)
    return "\n".join(out).strip()

def flex(q):
    """Regex for q tolerating any whitespace, including line breaks."""
    return re.compile(r"\s+".join(re.escape(w) for w in q.split()))

def flex_loose(q):
    """As flex, but tolerating MISSING whitespace and case.

    The source notes run words together in several places — "Startnimodipine",
    "oralnimodipine", "CRPreviewed—unremarkable", "Patinet consent". The old
    abridged cards had silently repaired all of them. The card must show the
    note as written, so the matcher bends instead of the text.
    """
    return re.compile(r"\s*".join(re.escape(w) for w in q.split()), re.I)

def js(s):
    """Escape for a double-quoted JS string literal."""
    return (s.replace("\\", "\\\\").replace('"', '\\"')
             .replace("\n", "\\n").replace("\r", ""))

# N6's highlight quoted text that is not in the note. The note says
# "Impression Suspected RCVS" — a SUSPECTED diagnosis, which the abridged
# card had rendered as a confirmed one. Anchor the real words instead.
OVERRIDE = {("N6", "Impression: Reversible cerebral vasoconstriction syndrome"):
            "Impression Suspected RCVS"}

report, unanchored = [], []
lines_out = []
for rid in sorted(OLD, key=lambda r: int(r[1:])):
    o, note = OLD[rid], FULL[rid]
    text = tidy(note["text"])
    placed, loose_used = 0, []
    for mk in o["marks"]:
        want = OVERRIDE.get((rid, mk), mk)
        hit = flex(want).search(text)
        if not hit:
            hit = flex_loose(want).search(text)
            if hit: loose_used.append(mk)
        if not hit:
            unanchored.append((rid, o["row"], mk))
            continue
        if "<mark>" in text[max(0, hit.start()-30):hit.start()]:
            continue
        text = text[:hit.start()] + "<mark>" + hit.group(0) + "</mark>" + text[hit.end():]
        placed += 1
    flag = "OVERRIDE" if any(k[0] == rid for k in OVERRIDE) else ""
    if loose_used: flag = (flag + " typo-tolerant").strip()
    report.append((rid, o["row"], len(o["raw"]), len(text), len(o["marks"]), placed, flag))
    lines_out.append(
        f' {rid}:["{o["ts"]}","{o["typ"]}",{o["row"]},"{js(text)}"],')
lines_out[-1] = lines_out[-1].rstrip(",")

print(f"{'rec':5}{'row':>5}{'was':>7}{'now':>7}{'marks':>7}{'placed':>7}  note")
print("=" * 62)
for rid, row, a, b, mk, pl, flag in report:
    warn = "" if pl == mk else f"  <-- {mk-pl} NOT PLACED"
    print(f"{rid:5}{row:>5}{a:>7}{b:>7}{mk:>7}{pl:>7}  {flag}{warn}")
tot_a = sum(r[2] for r in report); tot_b = sum(r[3] for r in report)
print("=" * 62)
print(f"card text {tot_a} -> {tot_b} chars   ({tot_b/tot_a:.1f}x)")
print(f"highlights {sum(r[4] for r in report)} -> placed {sum(r[5] for r in report)}")
if unanchored:
    print("\nCOULD NOT ANCHOR:")
    for rid, row, mk in unanchored:
        print(f"  {rid} row {row}: \"{mk[:70]}\"")

if not WRITE:
    print("\ndry run — pass --write to apply")
    sys.exit(0)

# ---- splice the new REC block in ----
a = src.index("const REC={")
# the object closes with "]}; at the END of the last record's own line, not on
# a line of its own — an earlier version searched for "\n};" and swallowed
# DOC, PRSB, SPANS, DRUG, GROUPS and drawTL along with it.
mend = re.search(r'^ N\d+:\[.*?\]\};[ \t]*$', src[a:], re.M)
assert mend, "could not find the end of the REC object"
b = a + mend.end()
new_src = src[:a] + "const REC={\n" + "\n".join(lines_out) + "\n};" + src[b:]

# ---- SPANS quote s6 must be verbatim ("Expresed", not "Expressed") ----
new_src = new_src.replace(
    '"Expressed understanding of her condition and asked questions about long-term "',
    '"Expresed understanding of her condition and asked questions about long-term "')

# ---- markUncarried must match across line breaks now ----
old_fn = re.search(r"function markUncarried\(r,text\)\{.*?\n\}", new_src, re.S).group(0)
new_fn = '''function markUncarried(r,text){
  /* the card now carries the note's own line structure, so a quote may span a
     line break — match on whitespace runs rather than on an exact substring */
  let out=text, n=0;
  SPANS.filter(x=>x[1]===r).forEach(([sid,,quote])=>{
    const q=quote.replace(/\\.$/,"").trim();
    const rx=new RegExp(q.split(/\\s+/)
      .map(w=>w.replace(/[.*+?^${}()|[\\]\\\\]/g,"\\\\$&")).join("\\\\s+"));
    const m=out.match(rx);
    if(m && out.indexOf("<mark>"+m[0]+"</mark>")<0){
      out=out.replace(rx,`<u class="nc">${m[0]}</u>`); n++; }
  });
  return {html:out,n};
}'''
new_src = new_src.replace(old_fn, new_fn)

# ---- the card must render the line structure ----
new_src = new_src.replace(
  ".evl .rowtx{display:block;background:var(--surface);border:1px solid var(--rule);",
  ".evl .rowtx{display:block;white-space:pre-wrap;max-height:340px;overflow-y:auto;"
  "background:var(--surface);border:1px solid var(--rule);")

SCREEN.write_text(new_src, encoding="utf-8")
print(f"\nwritten {SCREEN}")

def sub(label, old, new):

/* width and height from a PNG's IHDR chunk; no dependency needed for this */
function pngSize(buf) {
  if (buf.length < 24 || buf.readUInt32BE(0) !== 0x89504e47) return null;
  return { w: buf.readUInt32BE(16), h: buf.readUInt32BE(20) };
}

function figureBlock(n) {
  const file = pathmod.join(FIGDIR, `fig${n}.png`);
  if (!fs.existsSync(file)) return null;
  const data = fs.readFileSync(file);
  const size = pngSize(data);
  if (!size) return null;
  return new Paragraph({
    children: [new ImageRun({
      data,
      /* type is NOT optional. Without it docx writes the image as
         word/media/<hash>.undefined, and [Content_Types].xml has no Default for
         an "undefined" extension, so the .docx is a malformed OOXML package.
         Word opens it anyway and LibreOffice converts it, which is how it got
         past a render check; python-docx refuses it outright, which is how it
         was caught. Do not remove this line. */
      type: "png",
      transformation: { width: PAGE_W, height: Math.round(size.h * PAGE_W / size.w) },
    })],
    alignment: AlignmentType.CENTER, spacing: { before: 200, after: 60 },
  });
}

/* an appendix figure from any image file, placed at a chosen width so a tall
   photograph does not run off the page; numbered by the caller (A1, A2 ...) */
function FIGX(file, label, caption, widthPx) {
  const p = pathmod.join(FIGDIR, file);
  if (!fs.existsSync(p)) return [NOTE(`MISSING FIGURE ${label}: figures/${file}`)];
  const data = fs.readFileSync(p);
  let w = 0, h = 0;
  if (/\.png$/i.test(file)) { const s = pngSize(data); w = s.w; h = s.h; }
  else { /* JPEG: walk the markers to the SOF0/SOF2 frame header */
    let i = 2; while (i < data.length) {
      const marker = data[i + 1], len = data.readUInt16BE(i + 2);
      if (marker === 0xC0 || marker === 0xC2) { h = data.readUInt16BE(i + 5); w = data.readUInt16BE(i + 7); break; }
      i += 2 + len; } }
  return [
    new Paragraph({ children: [new ImageRun({ data, type: /\.png$/i.test(file) ? "png" : "jpg",
      transformation: { width: widthPx, height: Math.round(h * widthPx / w) } })],
      alignment: AlignmentType.CENTER, spacing: { before: 200, after: 60 } }),
    new Paragraph({ children: [new TextRun({ text: `Figure ${label}: ${caption}`, font: FONT, size: 19, italics: true })],
      alignment: AlignmentType.CENTER, spacing: { after: 200 } }),
  ];
}

function TABLE(widths, rows, opt = {}) {

          children: rich(line, { size: 19, bold: ri === 0 }),

}

# -*- coding: utf-8 -*-
"""
Check the built report for the mistakes a reader would catch.

Run after every build, on the clean copy (what a marker sees):

    node report_build.js --clean && python3 check_report.py

It exits non-zero if anything FAILS, so it can be trusted without reading it.
WARN lines are things that are expected while chapters are still being written
(a pointer to Section 3.6 before Chapter 3 exists) and must be clear by
submission.

WHAT IT CHECKS
  1. Cross-references. Every "Section 4.3", "§3.8", "Chapter 5", "Table B3",
     "Figure 6" and "Appendix B" must point at something that exists, and a
     section or appendix it points at must have text in it.
  2. Figures and tables. Numbered in order, each with a caption, each mentioned
     in the text at least once.
  3. Citations. Every author-year citation in the text has an entry in the
     References, and every entry is cited somewhere.
  4. Leftovers. No "TBC", "VERIFY", "[[", "[ FIGURE" or "CAPTURE:" in the clean
     copy; no phrases known to be wrong (list below).
  5. Style. Curly and straight double quotes mixed; dashes counted.
  6. Words. Chapter totals from wordcount.js against the 12,000 ceiling.
"""
import re, subprocess, sys
from pathlib import Path
import docx
from docx.table import Table
from docx.text.paragraph import Paragraph

HERE = Path(__file__).resolve().parent
DOCX = HERE / (sys.argv[1] if len(sys.argv) > 1 else "MSc_Project_Report_for_Mai.docx")

# phrases that were once in the report and are known to be wrong
STALE = {
    r"\bOstermann\b": "the 2025 study is Unnewehr et al., not Ostermann",
    r"1\.6 to 0\.8": "Haynes et al. (2009) report 1.5 to 0.8 per cent",
    r"ward and medical|ward & medical": "the timeline rows are emergency, medical teams, therapies",
    r"grouped for a single decision": "unused records are decided one by one",
    r"\bthree lanes\b": "the timeline has rows by service",
    r"Ragan et al\.,? \(?2015|Sacha et al\.,? \(?2015": "TVCG 22(1) is the 2016 issue",
}
NOT_AUTHORS = {"Table", "Tables", "Figure", "Figures", "Section", "Sections", "Chapter",
               "Appendix", "Episode", "January", "February", "March", "April", "May",
               "June", "July", "August", "September", "October", "November", "December",
               "Draft", "Version", "Standard", "Summer", "Autumn", "Spring", "Winter"}

fails, warns = [], []
def FAIL(m): fails.append(m)
def WARN(m): warns.append(m)

# ── read the document in order ──────────────────────────────────────────────
d = docx.Document(DOCX)
blocks = []          # (kind, style, text)
for el in d.element.body.iterchildren():
    tag = el.tag.split("}")[1]
    if tag == "p":
        p = Paragraph(el, d)
        try: st = p.style.name or ""
        except Exception: st = ""
        if p.text.strip(): blocks.append(("p", st, p.text.strip()))
    elif tag == "tbl":
        for row in Table(el, d).rows:
            blocks.append(("t", "", " | ".join(c.text.strip() for c in row.cells)))

def is_heading(st, lvl=None):
    return st.startswith("Heading") and (lvl is None or st == f"Heading {lvl}")

# sections: number -> has content?
sections, appendices, current = {}, {}, None
order = []
for kind, st, text in blocks:
    if kind == "p" and is_heading(st):
        m = re.match(r"^(\d+(?:\.\d+){0,2})\s", text)
        a = re.match(r"^Appendix ([A-G])\b", text)
        if m:   current = ("S", m.group(1)); sections.setdefault(m.group(1), 0)
        elif a: current = ("A", a.group(1)); appendices.setdefault(a.group(1), 0)
        else:   current = ("X", text)
        order.append(current)
        continue
    if current and current[0] == "S":
        sections[current[1]] += len(text.split())
    elif current and current[0] == "A":
        appendices[current[1]] += len(text.split())

def section_words(num):
    """words in a section including its subsections"""
    return sum(w for k, w in sections.items() if k == num or k.startswith(num + "."))

# References block
in_refs, refs = False, []
for kind, st, text in blocks:
    if kind == "p" and is_heading(st, 1):

    if a not in appendices: FAIL(f"xref: Appendix {a} does not exist")

def order_check(kind, caps_in_order):
    main = [c for c in caps_in_order if c.isdigit()]
    if [int(c) for c in main] != list(range(1, len(main) + 1)):
        FAIL(f"{kind}s out of order or missing: {main}")
    for letter in "ABCDEFG":
        sub = [int(c[1:]) for c in caps_in_order if c.startswith(letter)]
        if sub and sub != list(range(1, len(sub) + 1)):
            FAIL(f"{kind}s {letter} out of order: {sub}")
t_order = [m.group(1) for m in re.finditer(r"^Table ([A-G]?\d+)[.:] ", full, re.M)]

mentioned_t = {n for m in re.finditer(r"\bTables? ((?:[A-G]?\d+)(?:(?:, | and | to |–|-)[A-G]?\d+)*)", mention_text) for n in nums(m.group(1))}
mentioned_f = {n for m in re.finditer(r"\b(?:Figures?|Fig\.) ((?:[A-G]?\d+)(?:(?:, | and | to |–|-)[A-G]?\d+)*)", mention_text) for n in nums(m.group(1))}

for c in f_order:

    cites.add((name, year))
if re.search(r"STAKEHOLDER-APPLICATION", bodytext): cites.add(("STAKEHOLDER-APPLICATION", "n.d."))

known_names = {k[0] for k in ref_keys}
for name, year in sorted(cites):
    if (name, year) in ref_keys: continue
    if name in known_names:

    elif any(name == n for n, _ in ref_keys):

    else:

warns = list(dict.fromkeys(warns)); fails = list(dict.fromkeys(fails))
for w in warns: print("  WARN  " + w)
for f in fails: print("  FAIL  " + f)

}
